# BƯỚC 3: HUẤN LUYỆN NHÁNH TRUYỀN THỐNG (DEEPFM) CHO E-COMMERCE

Mô hình DeepFM sẽ chỉ học trên các đặc trưng danh mục có thật: `user_id`, `item_id`, `brand`, và `primary_category`. Các đặc trưng này được mã hóa nguyên bản từ dữ liệu gốc.

=== Cell 5 [code] ===
import torch
import torch.nn as nn
from torch.utils.data import Dataset, DataLoader
from sklearn.preprocessing import LabelEncoder
from sklearn.metrics import mean_squared_error, mean_absolute_error, roc_auc_score
import numpy as np
import pandas as pd

device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

# ==============================================================================
# 1. ĐỊNH NGHĨA LẠI CLASS DEEPFM VÀ HÀM ĐÁNH GIÁ (TRÁNH LỖI MẤT MEMORY)
# ==============================================================================
class DeepFM(nn.Module):
    def __init__(self, field_dims, embed_dim=16, mlp_dims=(64, 32), dropout=0.2):
        super().__init__()
        self.num_fields = len(field_dims)
        self.offsets = np.array((0, *np.cumsum(field_dims)[:-1]), dtype=np.longlong)
        total_vocab_size = sum(field_dims)
        
        self.linear = nn.Embedding(total_vocab_size, 1)
        self.bias = nn.Parameter(torch.zeros((1,)))
        
        self.embedding = nn.Embedding(total_vocab_size, embed_dim)
        nn.init.xavier_uniform_(self.embedding.weight)
        
        layers = []
        input_dim = self.num_fields * embed_dim
        for dim in mlp_dims:
            layers.append(nn.Linear(input_dim, dim))
            layers.append(nn.BatchNorm1d(dim))
            layers.append(nn.ReLU())
            layers.append(nn.Dropout(dropout))
            input_dim = dim
        layers.append(nn.Linear(input_dim, 1))
        self.mlp = nn.Sequential(*layers)

    def forward(self, x):
        x = x + x.new_tensor(self.offsets).unsqueeze(0)
        linear_part = torch.sum(self.linear(x), dim=1) + self.bias
        emb = self.embedding(x)
        square_of_sum = torch.sum(emb, dim=1) ** 2
        sum_of_square = torch.sum(emb ** 2, dim=1)
        fm_part = 0.5 * torch.sum(square_of_sum - sum_of_square, dim=1, keepdim=True)
        dnn_part = self.mlp(emb.view(-1, self.num_fields * emb.size(2)))
        out = linear_part + fm_part + dnn_part
        return out.squeeze(1)

def evaluate_metrics(model, data_loader):
    model.eval()
    preds, trues = [], []
    with torch.no_grad():
        for x, y in data_loader:
            x = x.to(device)
            pred = model(x).cpu().numpy()
            pred = np.clip(pred, 1.0, 5.0)
            preds.extend(pred)
            trues.extend(y.numpy())
            
    preds = np.array(preds)
    trues = np.array(trues)
    rmse = np.sqrt(mean_squared_error(trues, preds))
    mae = mean_absolute_error(trues, preds)
    binary_trues = (trues >= 4.0).astype(int)
    auc = roc_auc_score(binary_trues, preds)
    return rmse, mae, auc, preds

# ==============================================================================
# 2. CHUẨN BỊ ĐẶC TRƯNG VÀ HUẤN LUYỆN CHO AMAZON
# ==============================================================================
amazon_all_df = pd.concat([train_df, val_df, test_df], ignore_index=True)
amazon_all_df['primary_category'] = amazon_all_df['categories'].apply(lambda x: str(x).split(' | ')[0])

features = ['user_id', 'item_id', 'brand', 'primary_category']
field_dims = []
encoders = {}

for col in features:
    le = LabelEncoder()
    amazon_all_df[col] = le.fit_transform(amazon_all_df[col].astype(str))
    encoders[col] = le
    field_dims.append(amazon_all_df[col].nunique())

n_tr, n_v = len(train_df), len(val_df)
tr_enc = amazon_all_df.iloc[:n_tr].copy()
va_enc = amazon_all_df.iloc[n_tr:n_tr + n_v].copy()
te_enc = amazon_all_df.iloc[n_tr + n_v:].copy()

class RatingDataset(Dataset):
    def __init__(self, df, cols):
        self.x = df[cols].values.astype(np.int64)
        self.y = df['rating'].values.astype(np.float32)
    def __len__(self): return len(self.y)
    def __getitem__(self, idx): return self.x[idx], self.y[idx]

train_loader = DataLoader(RatingDataset(tr_enc, features), batch_size=512, shuffle=True)
val_loader   = DataLoader(RatingDataset(va_enc, features), batch_size=1024, shuffle=False)
test_loader  = DataLoader(RatingDataset(te_enc, features), batch_size=1024, shuffle=False)

deepfm = DeepFM(field_dims, embed_dim=16, mlp_dims=(64, 32), dropout=0.2).to(device)
criterion = nn.MSELoss()
optimizer = torch.optim.Adam(deepfm.parameters(), lr=0.001, weight_decay=1e-4)

best_val_rmse = float('inf')
final_test_preds = None

print("Đang huấn luyện DeepFM Backbone...")
for epoch in range(1, 6): # E-commerce data lớn, 5 epochs là đủ hội tụ
    deepfm.train()
    for x, y in train_loader:
        optimizer.zero_grad()
        loss = criterion(deepfm(x.to(device)), y.to(device))
        loss.backward()
        optimizer.step()
        
    val_rmse, _, _, _ = evaluate_metrics(deepfm, val_loader)
    test_rmse, test_mae, test_auc, test_preds = evaluate_metrics(deepfm, test_loader)
    
    if val_rmse < best_val_rmse:
        best_val_rmse = val_rmse
        final_test_preds = test_preds
        
    print(f"Epoch {epoch:02d} | Val RMSE: {val_rmse:.4f} | Test RMSE: {test_rmse:.4f} | Test MAE: {test_mae:.4f} | Test AUC: {test_auc:.4f}")

test_df['R_rec'] = final_test_preds
print("✅ Hoàn tất lưu R_rec từ mô hình truyền thống!")

import torch
from transformers import AutoTokenizer, AutoModelForCausalLM
from sklearn.feature_extraction.text import TfidfVectorizer

# Chỉ chạy phần tải mô hình nếu chưa tải
if 'llm_model' not in globals():
    print("Đang tải LLM Qwen-2.5-7B-Instruct...")
    MODEL_ID = "Qwen/Qwen2.5-7B-Instruct"
    tokenizer = AutoTokenizer.from_pretrained(MODEL_ID)
    llm_model = AutoModelForCausalLM.from_pretrained(MODEL_ID, torch_dtype=torch.float16, device_map="auto")

def ecom_generate_negative_desc(title, real_reviews):
    if not real_reviews: return "No significant negative feedback available."
    prompt = f"Here are some real low-rating reviews for the product '{title}':\n"
    for r in real_reviews: prompt += f"- \"{r}\"\n"
    prompt += f"Briefly summarize the main drawbacks or complaints about this product in 1-2 sentences."
    
    inputs = tokenizer.apply_chat_template([{"role": "user", "content": prompt}], tokenize=False, add_generation_prompt=True)
    in_tensors = tokenizer([inputs], return_tensors="pt").to(llm_model.device)
    with torch.no_grad():
        out = llm_model.generate(**in_tensors, max_new_tokens=100, temperature=0.1, do_sample=False)
    return tokenizer.batch_decode([out[0][len(in_tensors.input_ids[0]):]], skip_special_tokens=True)[0].strip()

def ecom_verify_kb(title, real_kb):
    # LLM sinh thông tin
    prompt = f"Provide the 'brand' and main 'category' for the beauty product '{title}'. Return in JSON format with keys 'brand', 'category'."
    inputs = tokenizer.apply_chat_template([{"role": "user", "content": prompt}], tokenize=False, add_generation_prompt=True)
    in_tensors = tokenizer([inputs], return_tensors="pt").to(llm_model.device)
    with torch.no_grad():
        out = llm_model.generate(**in_tensors, max_new_tokens=50, temperature=0.1, do_sample=False)
    text_out = tokenizer.batch_decode([out[0][len(in_tensors.input_ids[0]):]], skip_special_tokens=True)[0]
    
    try:
        llm_dict = json.loads(re.search(r'\{.*\}', text_out.replace('\n', '')).group(0))
    except:
        llm_dict = {"brand": "unknown", "category": "unknown"}
        
    verified = {}
    for field, kb_key in [('brand', 'brand'), ('category', 'categories')]:
        raw_val = str(llm_dict.get(field, "unknown")).lower()
        std_val = str(real_kb.get(kb_key, "unknown")).lower()
        
        if std_val == "unknown":
            verified[field] = raw_val; continue
            
        vec = TfidfVectorizer()
        try:
            tfidf = vec.fit_transform([raw_val, std_val])
            sim = cosine_similarity(tfidf[0:1], tfidf[1:2])[0][0]
            verified[field] = std_val if sim < 0.78 else raw_val
        except:
            verified[field] = std_val
    return verified

def build_ecom_cot_prompt(user_id, title, top_cor, top_neg, c_neg, kb_info):
    prompt = f"""You are a professional rating predictor for e-commerce recommendation systems. Predict the rating (1.0-5.0) that user {user_id} will give to the product '{title}'.
Verified Knowledge: Brand: {kb_info.get('brand')} | Category: {kb_info.get('category')}

I will provide information about 5 highest-rated products with similar categories this user bought:
{json.dumps(top_cor)}
And 5 lowest-rated products in their recent history:
{json.dumps(top_neg)}
The negative feedback summary for the target product is: {c_neg}

First, judge preference tendency based on historical behavior (Like: >=3, Dislike: <3).
Output strictly in this template:
user_id: {user_id}
product: {title}
preference: Like/Dislike
rating: x.x stars
reasons:
- Positive similarity: [Analyze]
- Negative similarity: [Analyze]"""
    return prompt

import numpy as np
import pandas as pd
import json
import re
from tqdm.auto import tqdm
from sklearn.metrics import mean_squared_error, mean_absolute_error, roc_auc_score
import torch
import os

# ==============================================================================
# 1. CÁC HÀM TRÍCH XUẤT (Giữ nguyên)
# ==============================================================================
def calc_sim_type(item_i, item_j, S, item_cat_indices):
    G_i = item_cat_indices.get(str(item_i), [])
    G_j = item_cat_indices.get(str(item_j), [])
    if not G_i or not G_j: return 0.0
    sub_matrix = S[np.ix_(G_i, G_j)]
    return float(np.mean(np.max(sub_matrix, axis=1)))

def extract_user_similarity_behavior(user_id, target_item_id, history_df, S, item_cat_indices, K=5):
    user_hist = history_df[history_df['user_id'] == user_id].copy()
    user_hist = user_hist[user_hist['item_id'] != target_item_id]
    if user_hist.empty: return [], []
    user_hist['sim_type'] = user_hist['item_id'].apply(lambda j: calc_sim_type(target_item_id, j, S, item_cat_indices))
    high_rated = user_hist[user_hist['rating'] >= 3].sort_values(by=['sim_type', 'rating'], ascending=[False, False]).head(K)
    top_cor = [{"product": row['title'], "category": row['categories'], "rating": row['rating']} for _, row in high_rated.iterrows()]
    low_rated = user_hist.sort_values(by=['rating', 'timestamp'], ascending=[True, False]).head(K)
    top_neg = [{"product": row['title'], "category": row['categories'], "rating": row['rating']} for _, row in low_rated.iterrows()]
    return top_cor, top_neg

def extract_score_from_cot(cot_text):
    match = re.search(r'rating:\s*([0-5](?:\.\d+)?)', cot_text, re.IGNORECASE)
    if match: return float(np.clip(float(match.group(1)), 1.0, 5.0))
    fallback = re.search(r'([1-5](?:\.\d+)?)\s*(?:stars|points)', cot_text, re.IGNORECASE)
    if fallback: return float(np.clip(float(fallback.group(1)), 1.0, 5.0))
    return 3.0 

# ==============================================================================
# 2. VÒNG LẶP ĐÁNH GIÁ CÓ LẤY MẪU VÀ CHECKPOINT
# ==============================================================================
alpha = 0.3
final_trues, final_preds_llm, final_preds_fin = [], [], []

test_df_sampled = test_df.sample(n=NUM_SAMPLES, random_state=42).reset_index(drop=True)

print(f"🚀 Bắt đầu đánh giá TrustRec trên {NUM_SAMPLES} mẫu Amazon E-commerce (đã lấy mẫu)...")
for idx, row in tqdm(test_df_sampled.iterrows(), total=len(test_df_sampled), desc="Inference"):
    u_id, i_id = str(row['user_id']), str(row['item_id'])
    title, r_true, r_rec = str(row['title']), float(row['rating']), float(row['R_rec'])
    
    top_cor, top_neg = extract_user_similarity_behavior(u_id, i_id, train_df, S_matrix_amazon, item_cat_indices, K=5)
    c_neg = ecom_generate_negative_desc(title, item_neg_reviews.get(i_id, []))
    c_sem = ecom_verify_kb(title, amazon_real_kb.get(i_id, {}))
    
    prompt = build_ecom_cot_prompt(u_id, title, top_cor, top_neg, c_neg, c_sem)
    inputs = tokenizer.apply_chat_template([{"role": "user", "content": prompt}], tokenize=False, add_generation_prompt=True)
    in_tensors = tokenizer([inputs], return_tensors="pt").to(llm_model.device)
    
    with torch.no_grad():
        out = llm_model.generate(**in_tensors, max_new_tokens=250, do_sample=False)
    out_text = tokenizer.batch_decode([out[0][len(in_tensors.input_ids[0]):]], skip_special_tokens=True)[0]
    
    r_llm = extract_score_from_cot(out_text)
    r_fin = alpha * r_llm + (1 - alpha) * r_rec
    
    final_trues.append(r_true)
    final_preds_llm.append(r_llm)
    final_preds_fin.append(r_fin)
    
    # CHỈNH SỬA: Tự động lưu checkpoint mỗi 50 mẫu để phòng ngắt kết nối
    if (idx + 1) % 50 == 0:
        temp_df = test_df_sampled.iloc[:idx+1].copy()
        temp_df['R_fin'] = final_preds_fin
        temp_df.to_csv('amazon_trustrec_checkpoint.csv', index=False)

# 3. KẾT QUẢ CUỐI CÙNG
# ==============================================================================
binary_trues = (np.array(final_trues) >= 4.0).astype(int)
print("\n🎯 KẾT QUẢ ĐÁNH GIÁ TẬP TEST AMAZON BEAUTY (TRUSTREC):")

print(f"   - MAE  : {mean_absolute_error(final_trues, final_preds_fin):.4f}")
print(f"   - AUC  : {roc_auc_score(binary_trues, final_preds_fin):.4f}")

    os.remove('amazon_trustrec_checkpoint.csv') # Xóa file tạm sau khi chạy xong
